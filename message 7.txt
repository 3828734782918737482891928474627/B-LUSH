-- Utility
local function Create(class, props, parent)
    local obj = Instance.new(class)
    for k,v in pairs(props) do
        obj[k] = v
    end
    obj.Parent = parent
    return obj
end

local function Round(obj, radius)
    Create("UICorner", {
        CornerRadius = UDim.new(0, radius)
    }, obj)
end

local function Stroke(obj, color, thickness)
    Create("UIStroke", {
        Color = color,
        Thickness = thickness or 1.5
    }, obj)
end

-- Sliders
local function Slider(name, min, max, initial, callback, format)
    local box = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 58),
        BackgroundColor3 = WHITE,
        BorderSizePixel = 0
    }, Scroll)

    Round(box, 10)
    Stroke(box, LIGHT, 1)

    local label = Create("TextLabel", {
        Size = UDim2.new(1, -20, 0, 22),
        Position = UDim2.fromOffset(10, 4),
        BackgroundTransparency = 1,
        TextColor3 = DARK,
        TextSize = 12,
        Font = Enum.Font.GothamBold,
        TextXAlignment = Enum.TextXAlignment.Left
    }, box)

    local bar = Create("TextButton", {
        Size = UDim2.new(1, -20, 0, 9),
        Position = UDim2.fromOffset(10, 37),
        BackgroundColor3 = LIGHT,
        Text = "",
        AutoButtonColor = false
    }, box)

    Round(bar, 10)

    local fill = Create("Frame", {
        BackgroundColor3 = PINK,
        Size = UDim2.new(0, 0, 1, 0),
        BorderSizePixel = 0
    }, bar)

    Round(fill, 10)

    local value = initial
    local dragging = false

    local function SetValue(v)
        value = math.clamp(v, min, max)

        if format == "int" then
            value = math.floor(value + 0.5)
        elseif format == "time" then
            value = math.floor(value * 100 + 0.5) / 100
        else
            value = math.floor(value * 10 + 0.5) / 10
        end

        local p = (value - min) / (max - min)
        fill.Size = UDim2.new(p, 0, 1, 0)
        label.Text = name .. "   •   " .. tostring(value)

        callback(value)
    end

    local function Update(x)
        local p = math.clamp(
            (x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X,
            0, 1
        )
        SetValue(min + (max - min) * p)
    end

    bar.MouseButton1Down:Connect(function()
        dragging = true
        Update(UIS:GetMouseLocation().X)
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            Update(input.Position.X)
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)

    SetValue(initial)
    return SetValue
end

-- Fog
local function ApplyFog()
    if FogEnabled then
        Lighting.FogStart = FogStart
        Lighting.FogEnd = math.max(
            FogStart + 1,
            FogStart + (FogEnd - FogStart) * (1 - FogStrength / 100)
        )
    else
        Lighting.FogStart = 0
        Lighting.FogEnd = 1000000
    end
end

-- Drag helper for both panel and toggle
local function MakeDraggable(object, handle)
    local dragging = false
    local startMouse
    local startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            startMouse = input.Position
            startPos = object.Position
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position-startMouse

            object.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset+delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset+delta.Y
            )
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging=false
        end
    end)
end
